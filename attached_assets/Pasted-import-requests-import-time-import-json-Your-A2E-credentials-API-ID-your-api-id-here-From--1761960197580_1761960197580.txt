# Your A2E credentials
API_ID = "your_api_id_here"  # From dashboard
API_KEY = "your_api_key_here"  # From dashboard

# Base URL (from docs; adjust if needed for image endpoint)
BASE_URL = "https://api.a2e.ai/"  # Or https://video.a2e.ai/ for specific gens

# Function for text-to-image (basic, no ref)
def generate_text_to_image(prompt, aspect_ratio="16:9"):
    payload = {
        "api_id": API_ID,
        "api_key": API_KEY,
        "text": prompt,  # Your detailed prompt
        "aspect_ratio": aspect_ratio,
        # Add params like style, resolution (e.g., "4k"), etc. from docs
    }
    response = requests.post(f"{BASE_URL}api/text_to_images/", json=payload)  # Endpoint inferred; check docs for exact
    if response.status_code == 200:
        job_id = response.json().get("id")
        return job_id
    else:
        print("Error:", response.text)
        return None

# Function for image editing (with ref image; upload ref URL or base64)
def edit_image_with_ref(ref_image_url, edit_prompt, aspect_ratio="16:9"):
    payload = {
        "api_id": API_ID,
        "api_key": API_KEY,
        "image_url": ref_image_url,  # URL to your reference image (upload to A2E first if needed)
        "text": edit_prompt,  # e.g., "Match exact face and pose, but add NSFW elements like lingerie"
        "aspect_ratio": aspect_ratio,
        # Optional: language, style (e.g., "photorealistic"), length if video
    }
    response = requests.post(f"{BASE_URL}api/image_edits/", json=payload)  # Use actual endpoint from docs, e.g., /api/image_to_videos/ for motion
    if response.status_code == 200:
        job_id = response.json().get("id")
        return job_id
    else:
        print("Error:", response.text)
        return None

# Poll for job status and result
def check_job_status(job_id, endpoint_type="images"):
    while True:
        response = requests.get(f"{BASE_URL}api/{endpoint_type}/{job_id}/", params={"api_id": API_ID, "api_key": API_KEY})
        if response.status_code == 200:
            data = response.json()
            status = data.get("status")
            if status == "completed":
                return data.get("output_url")  # Or "video_url" / "image_url"
            elif status == "failed":
                print("Job failed:", data.get("error"))
                return None
            print("Status:", status, "- Waiting 10s...")
            time.sleep(10)
        else:
            print("Error checking status:", response.text)
            return None
