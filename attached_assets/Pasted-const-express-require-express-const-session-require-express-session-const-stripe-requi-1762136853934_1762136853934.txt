const express = require('express');
const session = require('express-session');
const stripe = require('stripe')(process.env.STRIPE_SECRET_KEY);
const sqlite3 = require('sqlite3').verbose();
const app = express();

app.use(express.json());
app.use(express.static('public'));
app.use(session({
  secret: 'your-secret',
  resave: false,
  saveUninitialized: true
}));

// === DATABASE ===
const db = new sqlite3.Database('users.db');
db.serialize(() => {
  db.run(`CREATE TABLE IF NOT EXISTS users (
    id INTEGER PRIMARY KEY,
    email TEXT UNIQUE,
    credits INTEGER DEFAULT 0,
    tier TEXT,
    monthly_gens INTEGER DEFAULT 0,
    last_reset TEXT
  )`);
});

// === PRICE IDs (Replace with yours) ===
const SUB_PRICES = {
  basic: 'price_1UPDATED_BASIC',
  pro: 'price_1UPDATED_PRO',
  elite: 'price_1UPDATED_ELITE',
  unlimited: 'price_1SP8gJIUeudGFkNmOJCVnu6S'
};

// === HELPERS ===
function getUser(req, callback) {
  const email = req.session.email || 'guest@example.com';
  db.get(`SELECT * FROM users WHERE email = ?`, [email], (err, row) => {
    if (!row) {
      db.run(`INSERT INTO users (email, credits, tier) VALUES (?, 0, 'free')`, [email]);
      db.get(`SELECT * FROM users WHERE email = ?`, [email], (err, row) => callback(row));
    } else {
      callback(row);
    }
  });
}

function resetMonthly(user) {
  const now = new Date().toISOString().slice(0, 7); // YYYY-MM
  if (user.last_reset !== now) {
    db.run(`UPDATE users SET monthly_gens = 0, last_reset = ? WHERE id = ?`, [now, user.id]);
  }
}

// === ROUTES ===
app.get('/', (req, res) => {
  res.sendFile(__dirname + '/public/index.html');
});

app.post('/subscribe', async (req, res) => {
  const { tier } = req.body;
  const priceId = SUB_PRICES[tier];
  if (!priceId) return res.status(400).json({ error: 'Invalid tier' });

  try {
    const session = await stripe.checkout.sessions.create({
      payment_method_types: ['card'],
      line_items: [{ price: priceId, quantity: 1 }],
      mode: 'subscription',
      success_url: `${req.headers.origin}/success`,
      cancel_url: `${req.headers.origin}/cancel`,
      metadata: { email: req.session.email || 'guest@example.com', tier }
    });
    res.json({ url: session.url });
  } catch (e) {
    res.status(500).json({ error: e.message });
  }
});

app.post('/generate', (req, res) => {
  getUser(req, user => {
    resetMonthly(user);
    const isUnlimited = user.tier === 'unlimited';

    if (isUnlimited && user.monthly_gens >= 2000) {
      return res.status(429).json({ error: 'Monthly limit reached' });
    }

    if (!isUnlimited && user.credits <= 0) {
      return res.status(402).json({ error: 'No credits' });
    }

    if (!isUnlimited) {
      db.run(`UPDATE users SET credits = credits - 1 WHERE id = ?`, [user.id]);
    } else {
      db.run(`UPDATE users SET monthly_gens = monthly_gens + 1 WHERE id = ?`, [user.id]);
    }

    // === YOUR AI CALL HERE ===
    // e.g., call Eternal AI or fal.ai
    res.json({ image_url: 'https://via.placeholder.com/512' });
  });
});

app.get('/user', (req, res) => {
  getUser(req, user => {
    resetMonthly(user);
    res.json({
      credits: user.credits,
      tier: user.tier,
      monthly_gens: user.monthly_gens
    });
  });
});

// Webhook (optional)
app.post('/webhook', express.raw({type: 'application/json'}), (req, res) => {
  const sig = req.headers['stripe-signature'];
  let event;
  try {
    event = stripe.webhooks.constructEvent(req.body, sig, process.env.STRIPE_WEBHOOK_SECRET);
  } catch (err) { return res.status(400).send(`Webhook Error: ${err.message}`); }

  if (event.type === 'checkout.session.completed') {
    const session = event.data.object;
    const email = session.metadata.email;
    const tier = session.metadata.tier;
    db.run(`UPDATE users SET tier = ? WHERE email = ?`, [tier, email]);
  }
  res.json({ received: true });
});

app.listen(3000, () => console.log('Running on port 3000'));
