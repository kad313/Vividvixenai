  <script>
    async function sub(tier) {
      const res = await fetch('/subscribe', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ tier })
      });
      const data = await res.json();
      if (data.url) location = data.url;
    }

    async function generate() {
      const res = await fetch('/generate', { method: 'POST' });
      const data = await res.json();
      document.getElementById('result').innerHTML = data.image_url ? `<img src="${data.image_url}">` : `<p>${data.error}</p>`;
      updateUser();
    }

    async function updateUser() {
      const res = await fetch('/user');
      const user = await res.json();
      document.getElementById('credits').innerText = user.credits;
      document.getElementById('tier').innerText = user.tier;
    }
    updateUser();
    setInterval(updateUser, 5000);
  </script>
